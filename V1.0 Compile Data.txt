    <script>
        // Lưu trữ dữ liệu mẫu trên LocalStorage (Sau này có thể nâng cấp kết nối Supabase để lưu online cho nhiều người cùng nhập)
        let matches = JSON.parse(localStorage.getItem('compile_matches')) || [];

        const form = document.getElementById('matchForm');
        form.addEventListener('submit', function(e) {
            e.preventDefault();
            const newMatch = {
                id: Date.now(),
                p1Name: document.getElementById('p1Name').value.trim(),
                p1Protocols: document.getElementById('p1Protocols').value.split(',').map(s => s.trim()),
                p2Name: document.getElementById('p2Name').value.trim(),
                p2Protocols: document.getElementById('p2Protocols').value.split(',').map(s => s.trim()),
                winner: document.getElementById('winner').value
            };
            matches.push(newMatch);
            localStorage.setItem('compile_matches', JSON.stringify(matches));
            form.reset();
            renderApp();
        });

        function deleteMatch(id) {
            matches = matches.filter(m => m.id !== id);
            localStorage.setItem('compile_matches', JSON.stringify(matches));
            renderApp();
        }

        function renderApp() {
            const historyList = document.getElementById('matchHistoryList');
            const playerStatsDiv = document.getElementById('playerStats');
            const protocolStatsDiv = document.getElementById('protocolStats');

            if (matches.length === 0) {
                historyList.innerHTML = `<tr><td colspan="4" class="p-4 text-center text-slate-500">Chưa có trận đấu nào được ghi nhận.</td></tr>`;
                playerStatsDiv.innerHTML = `Chưa có dữ liệu.`;
                protocolStatsDiv.innerHTML = `Chưa có dữ liệu.`;
                return;
            }

            // Hiển thị lịch sử
            historyList.innerHTML = matches.slice().reverse().map(m => `
                <tr class="border-b border-slate-800 hover:bg-slate-800/50">
                    <td class="p-2"><strong>${m.p1Name}</strong><br><span class="text-xs text-slate-400">[${m.p1Protocols.join(', ')}]</span></td>
                    <td class="p-2"><strong>${m.p2Name}</strong><br><span class="text-xs text-slate-400">[${m.p2Protocols.join(', ')}]</span></td>
                    <td class="p-2 font-bold text-emerald-400">${m.winner === 'p1' ? m.p1Name : m.p2Name}</td>
                    <td class="p-2 text-right"><button onclick="deleteMatch(${m.id})" class="text-red-400 hover:text-red-300 text-xs bg-red-950/40 px-2 py-1 rounded border border-red-900">Xóa</button></td>
                </tr>
            `).join('');

            // Thống kê người chơi
            let playerRecords = {};
            let protocolRecords = {};

            matches.forEach(m => {
                let p1 = m.p1Name;
                let p2 = m.p2Name;
                
                if (!playerRecords[p1]) playerRecords[p1] = { win: 0, total: 0 };
                if (!playerRecords[p2]) playerRecords[p2] = { win: 0, total: 0 };
                
                playerRecords[p1].total++;
                playerRecords[p2].total++;

                if (m.winner === 'p1') playerRecords[p1].win++;
                else playerRecords[p2].win++;

                // Thống kê Protocol
                m.p1Protocols.forEach(proto => {
                    if (!protocolRecords[proto]) protocolRecords[proto] = { win: 0, total: 0 };
                    protocolRecords[proto].total++;
                    if (m.winner === 'p1') protocolRecords[proto].win++;
                });
                m.p2Protocols.forEach(proto => {
                    if (!protocolRecords[proto]) protocolRecords[proto] = { win: 0, total: 0 };
                    protocolRecords[proto].total++;
                    if (m.winner === 'p2') protocolRecords[proto].win++;
                });
            });

            playerStatsDiv.innerHTML = Object.keys(playerRecords).map(player => {
                let r = playerRecords[player];
                let percent = ((r.win / r.total) * 100).toFixed(1);
                return `<div class="flex justify-between items-center bg-slate-800/60 p-2 rounded mb-1">
                    <span>${player}</span>
                    <span class="text-xs font-mono"><strong class="text-emerald-400">${r.win}W</strong> / ${r.total}T (${percent}%)</span>
                </div>`;
            }).join('');

            protocolStatsDiv.innerHTML = Object.keys(protocolRecords).map(proto => {
                let r = protocolRecords[proto];
                let percent = ((r.win / r.total) * 100).toFixed(1);
                return `<div class="flex justify-between items-center bg-slate-800/60 p-2 rounded mb-1">
                    <span class="capitalize text-amber-200">${proto}</span>
                    <span class="text-xs font-mono"><strong class="text-emerald-400">${r.win}W</strong> / ${r.total}T (${percent}%)</span>
                </div>`;
            }).join('');
        }

        renderApp();
    </script>
